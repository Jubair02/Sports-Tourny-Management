     8→function hashPassword(password: string): string {
     9→  const salt = crypto.randomBytes(16).toString("hex");
    10→  const hash = crypto.scryptSync(password, salt, 64).toString("hex");
    11→  return `${salt}:${hash}`;
    12→}
